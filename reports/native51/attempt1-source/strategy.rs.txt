// Full one-wall exposure along every shortest-path branch. Geometry depends on
// walls only, so all pawn positions and both sides share its reverse BFS maps.
struct GoalGeometry {
    maps: [[i16; 81]; 2],
}

#[cfg(test)]
mod strategy_tests {
    use super::*;
    fn cache() -> SharedSearch {
        SharedSearch {
            profile: 2,
            ..SharedSearch::default()
        }
    }
    fn brute(p: &mut Position, target: usize) -> i32 {
        if p.left[1 - target] == 0 {
            return 0;
        }
        let base = p.distance(target, false);
        let other = p.distance(1 - target, false);
        let mut best = 0;
        for a in 81..209 {
            if !p.fits(a) {
                continue;
            }
            p.toggle(a, true);
            let d = p.distance(target, false);
            let e = p.distance(1 - target, false);
            p.toggle(a, false);
            if d < 100 && e < 100 {
                best = best.max(d - base - (e - other) - 1);
            }
        }
        best
    }
    #[test]
    fn full_threat_matches_exhaustive_walls_on_archived_losses() {
        let fixtures: serde_json::Value = serde_json::from_str(include_str!(concat!(
            env!("CARGO_MANIFEST_DIR"),
            "/../evaluation/fixtures/native-maze-regressions.json"
        )))
        .unwrap();
        let cache = cache();
        for f in fixtures.as_array().unwrap() {
            let state: State = serde_json::from_value(f["state"].clone()).unwrap();
            let mut p = Position::from_state(&state).unwrap();
            let key = p.key();
            let edges = p.edges;
            let target = f["target"].as_u64().unwrap() as usize;
            let found = cache.wall_threat(&mut p, target);
            assert_eq!(found.gain, brute(&mut p, target));
            assert_eq!(found.gain, f["gain"].as_i64().unwrap() as i32);
            assert!(p.key() == key);
            assert_eq!(p.edges, edges);
            if let Some(a) = found.action {
                assert!(found.delay >= 2);
                let old = p.turn;
                p.turn = 1 - target;
                assert!(p.actions().contains(&a));
                p.turn = old;
            }
            assert_eq!(
                cache.static_score(&mut p),
                cache.static_score(&mut mirror_position(&p))
            );
        }
    }
    #[test]
    fn cached_geometry_stays_correct_across_pawn_moves_and_illegal_enclosures() {
        let s:State=serde_json::from_str(r#"{"p":[{"x":0,"y":8},{"x":8,"y":0}],"walls":[{"x":0,"y":7,"o":"h"}],"left":[1,1],"turn":0,"winner":null}"#).unwrap();
        let mut p = Position::from_state(&s).unwrap();
        let cache = cache();
        for _ in 0..12 {
            for side in 0..2 {
                assert_eq!(cache.wall_threat(&mut p, side).gain, brute(&mut p, side));
            }
            let a = p.moves(p.turn)[0];
            p.make(a);
            if p.terminal() {
                break;
            }
        }
        p.left = [0, 0];
        assert_eq!(cache.wall_threat(&mut p, 0).gain, 0);
        assert_eq!(cache.wall_threat(&mut p, 1).gain, 0);
    }
    #[test]
    fn critical_defense_and_profitable_walls_keep_full_depth() {
        let s: State = serde_json::from_str(
            r#"{"p":[{"x":0,"y":5},{"x":4,"y":7}],"walls":[],"left":[1,1],"turn":0,"winner":null}"#,
        )
        .unwrap();
        let mut p = Position::from_state(&s).unwrap();
        let cache = cache();
        let actions = cache.ranked(&mut p);
        assert!(!actions.is_empty());
        for (a, _) in actions {
            assert!(cache.is_critical(&p, a));
        }
        let score = |n| {
            let mut p = p.clone();
            p.p = [76, 4];
            p.left = [n, 0];
            cache.static_score(&mut p)
        };
        assert!(score(1) - score(0) > score(10) - score(9));
    }
}
#[derive(Clone, Copy, Default)]
struct WallThreat {
    gain: i32,
    delay: i32,
    action: Option<u16>,
}
impl SharedSearch {
    fn geometry(&self, p: &mut Position, extra: Option<u16>) -> Arc<GoalGeometry> {
        let key = p.walls | extra.map_or(0, |a| 1u128 << (a - 81));
        let shard =
            ((key as u64 ^ (key >> 64) as u64).wrapping_mul(0x9e3779b97f4a7c15) >> 58) as usize;
        if let Some(v) = self.geometries[shard].read().unwrap().get(&key) {
            return v.clone();
        }
        if let Some(a) = extra {
            p.toggle(a, true);
        }
        let maps = [p.goal_map(0), p.goal_map(1)];
        if let Some(a) = extra {
            p.toggle(a, false);
        }
        let value = Arc::new(GoalGeometry { maps });
        let mut cache = self.geometries[shard].write().unwrap();
        if cache.len() >= 128 {
            cache.clear();
        }
        cache.insert(key, value.clone());
        value
    }
    fn wall_threat(&self, p: &mut Position, target: usize) -> WallThreat {
        let attacker = 1 - target;
        if p.left[attacker] == 0 {
            return WallThreat::default();
        }
        let base = self.geometry(p, None);
        let map = &base.maps[target];
        let distance = map[p.p[target]] as i32;
        let other = base.maps[attacker][p.p[attacker]] as i32;
        let mut candidates = [false; 128];
        let mut seen = [false; 81];
        let mut queue = vec![p.p[target]];
        seen[queue[0]] = true;
        for head in 0..81 {
            if head >= queue.len() {
                break;
            }
            let n = queue[head];
            for &q in &p.edges[n] {
                if q < 0 || map[q as usize] != map[n] - 1 {
                    continue;
                }
                let q = q as usize;
                if !seen[q] {
                    seen[q] = true;
                    queue.push(q);
                }
                let x = (n % 9).min(q % 9) as i32;
                let y = (n / 9).min(q / 9) as i32;
                let vertical = n / 9 == q / 9;
                let anchors = if vertical {
                    [(x, y - 1), (x, y)]
                } else {
                    [(x - 1, y), (x, y)]
                };
                for (x, y) in anchors {
                    if (0..8).contains(&x) && (0..8).contains(&y) {
                        let i = if vertical { 64 } else { 0 } + (y * 8 + x) as usize;
                        if p.fits((81 + i) as u16) {
                            candidates[i] = true;
                        }
                    }
                }
            }
        }
        let mut best = WallThreat::default();
        for (i, selected) in candidates.into_iter().enumerate() {
            if !selected {
                continue;
            }
            let a = (81 + i) as u16;
            let after = self.geometry(p, Some(a));
            let d = after.maps[target][p.p[target]] as i32;
            let e = after.maps[attacker][p.p[attacker]] as i32;
            if d < 0 || e < 0 {
                continue;
            }
            let delay = d - distance;
            let gain = delay - (e - other) - 1;
            if gain > best.gain {
                best = WallThreat {
                    gain,
                    delay,
                    action: Some(a),
                };
            }
        }
        best
    }
    fn static_score(&self, p: &mut Position) -> i32 {
        if self.profile == 0 {
            return p.evaluate();
        }
        let t = p.turn;
        let o = 1 - t;
        let (a, ar) = p.route(t);
        let (b, br) = p.route(o);
        let near = (p.p[0] % 9).abs_diff(p.p[1] % 9) + (p.p[0] / 9).abs_diff(p.p[1] / 9) <= 3;
        let da = if near { p.distance(t, true) } else { a };
        let db = if near { p.distance(o, true) } else { b };
        let attack = self.wall_threat(p, o);
        let defense = self.wall_threat(p, t);
        let stock = |n: usize| 120. * (n as f64).sqrt() + 24. * n as f64;
        let reserve = stock(p.left[t]) - stock(p.left[o]);
        let pressure = (attack.gain - defense.gain) * 80;
        (db - da) * 120
            + (b - a) * 12
            + if p.left[t] + p.left[o] > 0 {
                (ar - br) * 8
            } else {
                0
            }
            + (reserve + pressure as f64 + 0.5).floor() as i32
    }
    fn is_critical(&self, p: &Position, a: u16) -> bool {
        if self.profile < 2 {
            return false;
        }
        let (key, flip) = canonical(p.key());
        let a = if flip { mirror_action(a) } else { a };
        // If cache eviction removed the metadata, keep full depth conservatively.
        self.critical[Self::shard(&key)]
            .read()
            .unwrap()
            .get(&key)
            .is_none_or(|list| list.contains(&a))
    }
}
